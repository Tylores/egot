║                    WADL EXTRACTOR - QUICK REFERENCE                        ║
╚════════════════════════════════════════════════════════════════════════════╝

BUILD:
  go build -o bin/wadl-extract ./cmd/wadl-extract

SYNTAX:
  ./bin/wadl-extract -wadl <file> -path <prefix> -output <file> [-v]

BASIC USAGE:
  ./bin/wadl-extract -wadl wadl/sep_wadl.xml -path /dr -output dr.wadl

COMMON PATHS:
  /bill   - Billing resources (~10)
  /brs    - Billing reading sets (4)
  /dcap   - Device capability (1)
  /derp   - DER programs (6)
  /dr     - Demand response (5) ← Original use case
  /edev   - End devices (53)
  /file   - File management (2)
  /msg    - Messaging (5)
  /mup    - Mirror usage points (2)
  /ntfy   - Notifications (2)
  /ppy    - Prepayment (9)
  /rsps   - Response sets (3)
  /sdev   - Self device (1)
  /tm     - Time (1)
  /tp     - Tariff profiles (7)
  /upt    - Usage points (6)

FLAGS:
  -wadl string     Path to input WADL file (REQUIRED)
  -path string     Resource path prefix (REQUIRED)
  -output string   Output WADL file (REQUIRED)
  -v               Verbose mode (shows extracted resources)

EXAMPLES:
  # Extract demand-response
  ./bin/wadl-extract -wadl wadl/sep_wadl.xml -path /dr -output wadl/dr.wadl -v

  # Extract end-devices
  ./bin/wadl-extract -wadl wadl/sep_wadl.xml -path /edev -output wadl/edev.wadl

  # Extract messaging
  ./bin/wadl-extract -wadl wadl/sep_wadl.xml -path /msg -output wadl/msg.wadl

EXIT CODES:
  0 - Success
  1 - Error (invalid args, no matches, I/O error)

OUTPUT:
  ✓ Extracted WADL saved to: <path>
    Resources: <count>
    Path prefix: <prefix>

ERRORS:
  "failed to open WADL file"    → Check -wadl path
  "No resources found"          → Check -path prefix
  "failed to write WADL file"   → Check -output directory
